      if(!this.recording) return;

        const rms = Math.sqrt(sum/bufIn.length);
        const segments = 24; const segW = W/segments; const active = Math.floor(rms*segments*1.5);

  encodeWAV(buffer){
    const numCh = buffer.numberOfChannels;
    const len = buffer.length * numCh * 2 + 44;
    const arr = new ArrayBuffer(len);
    const view = new DataView(arr);
    const writeStr = (off, s)=>{ for(let i=0;i<s.length;i++) view.setUint8(off+i, s.charCodeAt(i)); };
    writeStr(0,'RIFF'); view.setUint32(4, len-8, true); writeStr(8,'WAVE');
    writeStr(12,'fmt '); view.setUint32(16, 16, true); view.setUint16(20, 1, true);
    view.setUint16(22, numCh, true); view.setUint32(24, buffer.sampleRate, true);
    view.setUint32(28, buffer.sampleRate*numCh*2, true); view.setUint16(32, numCh*2, true);
    view.setUint16(34, 16, true); writeStr(36,'data'); view.setUint32(40, len-44, true);
    let off = 44;
    const channels = [];
    for(let i=0;i<numCh;i++) channels.push(buffer.getChannelData(i));
    for(let i=0;i<buffer.length;i++){
      for(let c=0;c<numCh;c++){
        let s = Math.max(-1, Math.min(1, channels[c][i]));
        s = s < 0 ? s * 0x8000 : s * 0x7FFF;
        view.setInt16(off, s, true); off += 2;
      }
    }
    return new Blob([arr], {type:'audio/wav'});
  },
  
  encodeMP3(buffer, kbps){
    if(typeof lamejs === 'undefined') throw new Error('lamejs not loaded');
    const enc = new lamejs.Mp3Encoder(buffer.numberOfChannels, buffer.sampleRate, kbps);
    const blockLen = 1152;
    const mp3Data = [];
    if(buffer.numberOfChannels === 1){
      const mp3buf = enc.encodeBuffer(buffer.getChannelData(0));
      if(mp3buf.length > 0) mp3Data.push(mp3buf);
    } else {
      const left = buffer.getChannelData(0);
      const right = buffer.getChannelData(1);
      for(let i=0;i<left.length;i+=blockLen){
        const l = left.subarray(i, i+blockLen);
        const r = right.subarray(i, i+blockLen);
        const mp3buf = enc.encodeBuffer(l, r);
        if(mp3buf.length > 0) mp3Data.push(mp3buf);
      }
    }
    const end = enc.flush();
    if(end.length > 0) mp3Data.push(end);
    return new Blob(mp3Data, {type:'audio/mp3'});
  },
  
  download(blob, name, type){
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url; a.download = name; a.click();
    setTimeout(()=>URL.revokeObjectURL(url), 1000);
  },
  
  formatTime(ms){
    const s = Math.floor(ms/1000);
    const m = Math.floor(s/60);
    const sec = s % 60;
    return `${String(m).padStart(2,'0')}:${String(sec).padStart(2,'0')}`;
  }

function drawHeroBars(){
  const canvas = document.getElementById('hero-canvas');
  if(!canvas) return;
  const ctx = canvas.getContext('2d');
  const W = canvas.width = window.innerWidth;
  const H = canvas.height = window.innerHeight;
  const bars = 80; const barW = W / bars; let t = 0;
  function draw(){
    t += 0.02; ctx.clearRect(0,0,W,H);
    for(let i=0; i<bars; i++){
      const phase = i * 0.15;
      const h = (Math.sin(t + phase) * 0.5 + Math.sin(t*1.3 + phase*1.7) * 0.3 + Math.random()*0.1) * H * 0.4 + 20;
      const x = i * barW; const y = H/2 - h/2;
      const grad = ctx.createLinearGradient(0, y, 0, y+h);
      grad.addColorStop(0, 'rgba(255,45,122,0.7)');
      grad.addColorStop(0.5, 'rgba(155,92,255,0.5)');
      grad.addColorStop(1, 'rgba(0,229,255,0.7)');
      ctx.fillStyle = grad;
      ctx.fillRect(x+1, y, barW-2, h);
    }
    requestAnimationFrame(draw);
  }
  draw();
}

function drawHeroWave(){
  const canvas = document.getElementById('hero-wave');
  if(!canvas) return;
  const ctx = canvas.getContext('2d');
  canvas.width = canvas.offsetWidth * 2;
  canvas.height = 240;
  const W = canvas.width, H = canvas.height;
  let t = 0;
  function draw(){
    t += 0.04; ctx.clearRect(0,0,W,H);
    ctx.strokeStyle = 'rgba(255,45,122,0.8)'; ctx.lineWidth = 3; ctx.beginPath();
    for(let i=0;i<W;i+=2){
      const y = H/2 + Math.sin(i*0.02 + t) * 30 + Math.sin(i*0.05 + t*1.3) * 15 + (Math.random()-0.5)*8;
      if(i===0) ctx.moveTo(i,y); else ctx.lineTo(i,y);
    }
    ctx.stroke();
    ctx.strokeStyle = 'rgba(0,229,255,0.3)'; ctx.lineWidth = 8; ctx.beginPath();
    for(let i=0;i<W;i+=2){
      const y = H/2 + Math.sin(i*0.02 + t) * 30 + Math.sin(i*0.05 + t*1.3) * 15;
      if(i===0) ctx.moveTo(i,y); else ctx.lineTo(i,y);
    }
    ctx.stroke();
    requestAnimationFrame(draw);
  }
  draw();
}

function drawPreviewWave(){
  const canvas = document.getElementById('preview-wave');
  if(!canvas) return;
  const ctx = canvas.getContext('2d');
  canvas.width = canvas.offsetWidth * 2;
  canvas.height = 160;
  const W = canvas.width, H = canvas.height;
  let t = 0;
  function draw(){
    t += 0.03; ctx.clearRect(0,0,W,H);
    const grad = ctx.createLinearGradient(0,0,W,0);
    grad.addColorStop(0,'rgba(255,45,122,0.9)');
    grad.addColorStop(0.5,'rgba(155,92,255,0.9)');
    grad.addColorStop(1,'rgba(0,229,255,0.9)');
    ctx.fillStyle = grad;
    const bars = 60; const bw = W / bars;
    for(let i=0;i<bars;i++){
      const h = Math.abs(Math.sin(i*0.3 + t) * 0.6 + Math.sin(i*0.7 + t*1.5) * 0.4) * H * 0.7;
      ctx.fillRect(i*bw+2, H/2-h/2, bw-3, h);
    }
    requestAnimationFrame(draw);
  }
  draw();
}
